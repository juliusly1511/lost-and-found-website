-----------------------------------------------
#DETAILEDD DATABASE STEP BY STEP#
----------------------------------------------------
:: 1. Open Command Prompt as Administrator
:: 2. Navigate to PostgreSQL bin directory
cd "C:\Program Files\PostgreSQL\16\bin"

:: 3. Create database and user
psql -U postgres -c "CREATE DATABASE lost_and_found;"
psql -U postgres -c "CREATE USER lost_found_user WITH PASSWORD 'your_password';"
psql -U postgres -c "GRANT ALL PRIVILEGES ON DATABASE lost_and_found TO lost_found_user;"

:: 4. Run the schema
psql -U postgres -d lost_and_found -f C:\path\to\your\project\database\schema.sql





------------------------------------------
BACKEND SERVER SETUP
-------------------------------------------
:: 1. Open Command Prompt
:: 2. Navigate to server directory
cd C:\path\to\your\project\server

:: 3. Install dependencies
npm install

:: 4. Create .env file manually or using command:
echo DB_HOST=localhost > .env
echo DB_PORT=5432 >> .env
echo DB_NAME=lost_and_found >> .env
echo DB_USER=lost_found_user >> .env
echo DB_PASSWORD=your_password >> .env
echo JWT_SECRET=your_super_secret_jwt_key_here >> .env
echo PORT=5000 >> .env

:: 5. Start the development server
npm run dev

:: Or for production:
npm start







------------------------------------------
FRONTEND CLIENT SETUP
------------------------------------------
:: 1. Open a NEW Command Prompt
:: 2. Navigate to client directory
cd C:\path\to\your\project\client

:: 3. Install dependencies
npm install

:: 4. Start the development server
npm run dev

:: 5. The app will open in your browser at http://localhost:3000







--------------------------------------------------------------------------
CREATE A BATCH File
create a file called setup.bat in root folder
-----------------------------------------------------------------------------
@echo off
echo Setting up Lost and Found Application...

echo Step 1: Setting up backend...
cd server
call npm install
if %errorlevel% neq 0 (
    echo Backend installation failed!
    pause
    exit /b 1
)

echo Step 2: Setting up frontend...
cd ..\client
call npm install
if %errorlevel% neq 0 (
    echo Frontend installation failed!
    pause
    exit /b 1
)





echo.
echo ========================================
echo Setup completed successfully!
echo.
echo To start the application:
echo   1. Backend: cd server && npm run dev
echo   2. Frontend: cd client && npm run dev
echo.
echo Make sure PostgreSQL is running and .env is configured.
echo ========================================
pause






---------------------------------------------------------------------------
START APPLICATION BATCH File
create start-app.bat
---------------------------------------------------------------------------
@echo off
echo Starting Lost and Found Application...

echo Starting backend server...
start cmd /k "cd server && npm run dev"

timeout /t 5

echo Starting frontend development server...
start cmd /k "cd client && npm run dev"

echo Both servers are starting...
echo Backend: http://localhost:5000
echo Frontend: http://localhost:3000
pause

---------------------------------------------------------------------------
TROUBLESHOOTING
if postgressql commandds dont work
----------------------------------------------------------------------------
:: Add PostgreSQL to PATH temporarily
set PATH=%PATH%;C:\Program Files\PostgreSQL\16\bin

:: Or use full path to psql
"C:\Program Files\PostgreSQL\16\bin\psql.exe" -U postgres -c "your command"


----------------------------------------------------------------------------
IF PORTS ALREADY IN USE :
--------------------------------------------------------------------------
:: Find what's using port 5000 (backend)
netstat -ano | findstr :5000

:: Find what's using port 3000 (frontend)  
netstat -ano | findstr :3000

:: Kill process by PID
taskkill /PID [PID_NUMBER] /F


-------------------------------------------------------------------------
IF NPM COMMANDS failed
----------------------------------------------------------------------
:: Clear npm cache
npm cache clean --force

:: Delete node_modules and reinstall
rmdir /s node_modules
npm install

PRODUCTION DEPLOYMENT NOTES:
For production deployment, you'll want to:
----------------------------------------------------------------------
1.build frontend:
cd client
npm run build

2.serve front end from backend 
add this to server/server.js:


=====================================================
import path from 'path';
import { fileURLToPath } from 'url';

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

// Serve static files from React build
app.use(express.static(path.join(__dirname, '../client/dist')));

// Handle React routing, return all requests to React app
app.get('*', (req, res) => {
  res.sendFile(path.join(__dirname, '../client/dist/index.html'));
});


===================================================================
3. use enviroment specific variables 
:: For production
set NODE_ENV=production
